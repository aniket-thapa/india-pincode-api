Quick Start

Up in 30 seconds

Copy any snippet below and you're already fetching real postal data. 

// Pincode lookup — works in browser or Node.js
const BASE = 'https://aniket-thapa.github.io/india-pincode-api';

async function lookupPincode(pin) {
  const res = await fetch(`${BASE}/pincodes/${pin}.json`);
  if (!res.ok) throw new Error('Pincode not found');
  return res.json();
}

const data = await lookupPincode('110001');
console.log(data.state);    // "DELHI"
console.log(data.district); // "NEW DELHI"
console.log(data.offices.length); // number of offices

// Filter delivery-only offices with coordinates
const mapped = data.offices.filter(
  o => o.deliveryStatus === 'Delivery' && o.latitude !== null
);

// React hook example
import { useState, useEffect } from 'react';

const BASE = 'https://aniket-thapa.github.io/india-pincode-api';

function usePincode(pin) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    if (!pin) return;
    setLoading(true);
    fetch(`${BASE}/pincodes/${pin}.json`)
      .then(r => r.json())
      .then(setData)
      .catch(setError)
      .finally(() => setLoading(false));
  }, [pin]);

  return { data, loading, error };
}

Returns every post office in a district — with name, type, delivery status, circle, region, division, pincode, and GPS coordinates where available.

GET /pincodes/{6-digit-pincode}.json Pincode lookup ▼

The most commonly used endpoint. Returns all post offices under a specific 6-digit pincode, with state and district context. The pincode field is omitted in the office objects since it is already encoded in the filename.

💡
Example: /pincodes/400001.json (Mumbai GPO), /pincodes/110001.json (Connaught Place)
⚠️
Note: A single pincode often maps to multiple offices — always iterate offices[]. Always null-check latitude and longitude before using coordinates.
Sample Response

Field Reference

Every field explained

Office objects returned from endpoints share a consistent schema.

Field Type Description
officeName string Full name of the post office or branch office
officeType string HO (Head Office) · SO (Sub Office) · BO (Branch Office)
deliveryStatus string "Delivery" or "Non Delivery" — whether the office handles mail delivery
circleName string Postal circle (e.g. "Rajasthan Circle", "Maharashtra Circle")
regionName string Postal region within the circle
divisionName string Postal division within the region
latitude number | null WGS-84 latitude. null when not available in source data
longitude number | null WGS-84 longitude. null when not available in source data
pincode string 6-digit pincode. District endpoint only — omitted in pincode endpoint responses
Slug Convention

State and district names are converted to lowercase, hyphen-separated slugs. Rules: lowercase → replace non-alphanumeric sequences with - → trim hyphens.

Official Name Slug
UTTAR PRADESH uttar-pradesh
JAMMU AND KASHMIR jammu-and-kashmir
KUMURAM BHEEM ASIFABAD kumuram-bheem-asifabad
THE DADRA AND NAGAR HAVELI AND DAMAN AND DIU the-dadra-and-nagar-haveli-and-daman-and-diu

pincode-validator.js 
const BASE = 'https://aniket-thapa.github.io/india-pincode-api';

// Validate & enrich a pincode
async function validatePin(pin) {
  try {
    const { state, district, offices }
      = await fetch(`${BASE}/pincodes/${pin}.json`)
        .then(r => r.ok ? r.json() : null);
    return {
      valid: true, state, district,
      officeCount: offices.length,
      hasDelivery: offices.some(
        o => o.deliveryStatus === 'Delivery'
      )
    };
  } catch {
    return { valid: false };
  }
}

district_offices.py 
import requests

BASE = "https://aniket-thapa.github.io/india-pincode-api"

def get_mapped_offices(state, district):
    """Return offices with valid GPS coordinates."""
    url = f"{BASE}/districts/{state}/{district}.json"
    data = requests.get(url).json()
    return [
        o for o in data["offices"]
        if o["latitude"] is not None
    ]

offices = get_mapped_offices(
    "rajasthan", "jodhpur"
)
print(f"Mapped: {len(offices)} offices")

Python
bulk_lookup.py 
import requests, time

def bulk_lookup(pincodes, delay=0.2):
    """Lookup multiple pincodes with polite delay."""
    results = {}
    for pin in pincodes:
        try:
            r = requests.get(
                f"{BASE}/pincodes/{pin}.json",
                timeout=5
            )
            results[pin] = r.json() if r.ok else None
        except:
            results[pin] = None
        time.sleep(delay)
    return results

data = bulk_lookup(["110001", "400001", "600001"])
